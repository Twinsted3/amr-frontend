cmake_minimum_required(VERSION 3.16)
project(warehouse_safety)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim9 REQUIRED)
add_library(warehouse_command_watchdog SHARED src/command_watchdog.cc)
target_compile_features(warehouse_command_watchdog PUBLIC cxx_std_17)
target_link_libraries(warehouse_command_watchdog gz-sim9::gz-sim9)
install(TARGETS warehouse_command_watchdog LIBRARY DESTINATION lib)
ament_package()
